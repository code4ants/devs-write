{{- /* Theme of a page: a story's own `theme`, inherited by its chapters. Only stories (content/stories//) and their chapters have one; home, author, challenge and contribution pages do not. A chapter is a page whose parent section is a story, so stray files in an author folder (e.g. a git-ignored `local/` draft) get no theme. */ -}} {{- $themes := slice "fantasy" "sf" "newspaper" "dusty" -}} {{- /* Challenge pages (list, challenge, contributions) always use the "challenge" theme; it is not selectable by stories. */ -}} {{- if eq .Section "challenges" -}} {{- return "challenge" -}} {{- end -}} {{- $story := false -}} {{- if eq .Section "stories" -}} {{- if and (eq .Kind "section") (eq (len (split (strings.Trim .Path "/") "/")) 3) -}} {{- $story = . -}} {{- else if and (eq .Kind "page") (eq (len (split (strings.Trim .Parent.Path "/") "/")) 3) -}} {{- $story = .Parent -}} {{- end -}} {{- end -}} {{- $theme := "" -}} {{- if $story -}} {{- $theme = lower (string $story.Params.theme) -}} {{- if not (in $themes $theme) -}} {{- warnf "%s: theme %q invalid, using fantasy" $story.File.Path $story.Params.theme -}} {{- $theme = "fantasy" -}} {{- end -}} {{- end -}} {{- return $theme -}}
